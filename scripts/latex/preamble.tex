% =====================================================================
%  Print design for the C64 Commander / C64U Remote manuals.
%
%  The reference is the 1982 Commodore 64 User's Guide: a book a reader
%  settles into, not a datasheet. Three decisions carry most of the work.
%
%  1. A picture never floats. In a manual, a screenshot IS the sentence
%     before it; letting LaTeX defer it to the next page, or stack four
%     of them on one, breaks the instruction it belongs to. Every figure
%     is set exactly where it was written.
%  2. Every chapter owns a hue, and that hue is everywhere in it: the
%     opener, the running head, the section numbers, the figure labels,
%     the table head, the bullets, the callouts, and a thumb tab on the
%     outer edge of every page. Riffle the book and you can see where
%     you are.
%  3. Guidance that the prose already carries - what a feature needs
%     before it will work, which of two routes to prefer - is lifted out
%     of the paragraph and set as a callout, so it can be found by
%     someone skimming.
% =====================================================================

%  The book is set in IBM Plex, the family the app itself uses, so the printed
%  page and the screen are one design.
%
%  The faces are vendored under `fonts/` and loaded by path rather than by name.
%  Two reasons. A font installed on the machine is a font that can be absent or
%  a different version on someone else's, and the manual would then set in
%  whatever fontconfig picked without saying so. And loading by name needs a
%  fontconfig database, which a minimal TeX Live in CI does not have.
%
%  %%FONTDIR%% is replaced with an absolute path by build-manuals-latex.mjs.
% Pandoc 3.11 wraps every image in \pandocbounded, which its own template
% defines and a hand-written preamble does not. It exists to shrink an oversized
% picture into the text block; this build already sizes each screenshot from the
% shape of its file, so the identity is right. Declared with \providecommand so
% an older pandoc, which does not emit it, is unaffected.
\providecommand{\pandocbounded}[1]{#1}

% ----------------------------------------------------------- thumb tabs
% A printed manual's thumb index: one tab per chapter, stepped down the
% outer edge, filled for the chapter you are in and ghosted for the rest.
% Riffling the closed book shows the colour bands; a page on its own tells
% you which chapter it came from without reading a word.
\newcounter{tabloop}
\newlength{\tabheight}\setlength{\tabheight}{17mm}
\newlength{\tabwidth}\setlength{\tabwidth}{7mm}
\newlength{\tabtop}\setlength{\tabtop}{48mm}
\newcommand{\thumbcolour}[1]{%
  \ifcase#1\relax
    chapA\or chapA\or chapB\or chapC\or chapD\or chapE\or chapF\or chapG\or chapH\or chapI\else chapA\fi}
% Every tab is flush with the outer edge, on both sides of the spread, so the
% band for the chapter you are in is visible on the closed book. A tab that is
% not the current chapter is shorter, and gives up its width on the inner side;
% insetting it from the edge instead left it floating clear of the paper edge
% and invisible once the book was shut.
\newlength{\ghosttabwidth}\setlength{\ghosttabwidth}{\dimexpr\tabwidth-2.6mm\relax}
\newcommand{\drawthumbtabs}{%
  \ifnum\value{chapter}>0\relax
  \setcounter{tabloop}{0}%
  \loop\ifnum\value{tabloop}<9\relax
    \stepcounter{tabloop}%
    \edef\tabhue{\thumbcolour{\value{tabloop}}}%
    \ifnum\value{tabloop}=\value{chapter}\relax
      \edef\thistabwidth{\the\tabwidth}%
    \else
      \edef\thistabwidth{\the\ghosttabwidth}%
    \fi
    % The outer edge is the right on a recto and the left on a verso, so the
    % origin moves but the tab always ends at the paper edge. Both branches go
    % through LenToUnit: a bare dimension here shifted the verso tabs off the
    % edge by about a millimetre, which is what made the two sides look uneven.
    \ifodd\value{page}%
      \put(\LenToUnit{\dimexpr\paperwidth-\thistabwidth\relax},%
           \LenToUnit{\dimexpr-\tabtop-\value{tabloop}\tabheight\relax}){%
        \ifnum\value{tabloop}=\value{chapter}\relax
          \textcolor{\tabhue}{\rule{\thistabwidth}{\tabheight}}%
        \else
          \textcolor{\tabhue!16!c64paper}{\rule{\thistabwidth}{\tabheight}}%
        \fi}%
    \else
      \put(\LenToUnit{0pt},%
           \LenToUnit{\dimexpr-\tabtop-\value{tabloop}\tabheight\relax}){%
        \ifnum\value{tabloop}=\value{chapter}\relax
          \textcolor{\tabhue}{\rule{\thistabwidth}{\tabheight}}%
        \else
          \textcolor{\tabhue!16!c64paper}{\rule{\thistabwidth}{\tabheight}}%
        \fi}%
    \fi
  \repeat\fi}
\AddToShipoutPictureFG{\AtPageUpperLeft{\drawthumbtabs}}

% A figure never floats.
%
% In a step-by-step guide the screenshot IS the sentence above it. Letting
% LaTeX place it defers the picture to wherever it next fits: the reader meets
% it at the top of a page whose instruction is overleaf, and two or three of
% them stacked together after that. Both were tried here, with `[!ht]`, tuned
% float fractions and a `\FloatBarrier` at every section; the pictures still
% drifted, because four screenshots inside one section simply cannot all fit
% beside their own text.
%
% So the picture is set exactly where it was written, and the price is paid in
% white space at the foot of the occasional page. That is the cheaper mistake:
% a short page costs a reader nothing, a picture attached to the wrong sentence
% costs them the instruction.
%
% A `minipage` rather than `\needspace`: a minipage cannot be split, so LaTeX
% breaks the page before it whenever it will not fit whole. `\needspace` only
% reserves a fixed amount, and a screenshot taller than the reservation ran off
% the foot of the page with its caption lost.
\newenvironment{manualfigure}
  {\par\vspace{0.5\baselineskip}\noindent\begin{minipage}{\linewidth}\centering}
  {\end{minipage}\par\vspace{0.35\baselineskip}}
